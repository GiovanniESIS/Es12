Dato un array { name, scores: [] } 👉 Restituisci { name: media } 
  


    const user = [
        { name: "Giovanni", scores: [1, 2, 3] },
        { name: "Federico", scores: [2, 3, 4] }
    ];

    const transform = user.map(item => {
        const sum = item.scores.reduce((acc, number) => acc + number, 0);
        const media = sum / item.scores.length;
        return { name: item.name, media };
    });

    console.log(transform);

Dato un array { name, salary } 👉 Calcola la somma degli stipendi > 1000

        {name: 'Giovanni', salary:1468},
        {name: 'Andrei', salary: 5800},
        {name: 'federico', salary: 940}
    ]

    const sum = users.reduce((acc,item) => {

            return acc+item.salary
        }
        return acc
    },0)

    console.log(sum)
